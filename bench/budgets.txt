# Release budgets for macOS, checked by `choscorui-bench --check bench/budgets.txt`.
# Reference machine and baselines: docs/performance.md. Each limit sits above
# the measured p95 by a stated margin, or at a user-visible goal (a 60 Hz
# frame is 16.7 ms; a 30 Hz frame is 33 ms), whichever is tighter.
# Frames that move content stay inside one 120 Hz frame (8.3 ms), measured
# to the time the frame is on the display.
# Correctness counters must stay at zero.

# Startup: process entry to first presented frame.
startup.gallery.first_frame_ms.p95 <= 250
startup.stress.first_frame_ms.p95 <= 400
startup.gallery.failures <= 0
startup.stress.failures <= 0

# Idle: no repeated drawing, negligible CPU and wakeups.
idle.paints <= 0
idle.cpu_percent <= 1
idle.wakeups_per_sec <= 20
idle.footprint_mib.after <= 200

# Resize between 400x300 and 1600x1000 points.
resize.present_ms.p95 <= 8.3
resize.missed_configure <= 0
resize.missed_present <= 0

# 100,000-row list and 20,000-item canvas, wheel scrolling.
scroll.present_ms.p95 <= 8.3
scroll.frame_interval_ms.p95 <= 8.3
scroll.peak_realized_rows <= 40
scroll.allocations_per_frame.p95 <= 10000
scroll.missed_present <= 0

# Typing in a 100,000-character multilingual document.
document.present_ms.p95 <= 33
document.shape_ms.p95 <= 12
document.caret_mismatches <= 0
document.missed_present <= 0

# Structurally distinct light/dark theme switch, 240 controls.
theme.present_ms.p95 <= 8.3
theme.missed_present <= 0

# One state change in a 10,000-row view tree. The changed-node count must
# not grow with the tree; the frame stays inside a 120 Hz frame.
views.present_ms.p95 <= 8.3
views.dirty_nodes <= 4
views.dirty_growth <= 0
views.missed_present <= 0
views.errors <= 0

# Semantics of large models: 10,000 commands, 100,000 list rows, and a
# 20,000-item canvas. Only the parts in view publish, so the entries and
# the time of a key step and a semantics read do not grow with the model.
semantics.command_ms.p95 <= 1
semantics.list_ms.p95 <= 1
semantics.canvas_ms.p95 <= 1
semantics.command_entries <= 40
semantics.list_entries <= 60
semantics.canvas_entries <= 1000
semantics.command_moved >= 60
semantics.list_moved >= 60
semantics.canvas_moved >= 60
semantics.command_errors <= 0
semantics.list_errors <= 0
semantics.canvas_errors <= 0

# Open, resize, and close 20 windows twice.
windows.first_frame_ms.p95 <= 33
windows.leaked_native_windows <= 0
windows.missed_present <= 0
windows.idle_cpu_percent_after_close <= 3

# Release executable size, as the release profile builds it: the
# measured size plus one 16 KiB page. The cap is 3.25 MiB. The Metal
# painter adds GPU painting for 120 Hz frames. The component behavior
# fixes add control states, semantic parts, and gallery demos.
size.choscorui-gallery_mib <= 3.147
